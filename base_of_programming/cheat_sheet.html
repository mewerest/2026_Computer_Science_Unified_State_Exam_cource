1. Переменные, типы данных и операции

Типы данных

# int — целые числа
x = 42

# float — числа с плавающей точкой
pi = 3.14

# str — строки (текст)
name = "Анна"

# bool — логический тип
flag = True   # или False

Арифметические операции

a + b    # сложение
a - b    # вычитание
a * b    # умножение
a ** b   # возведение в степень
a / b    # деление (результат — float)
a // b   # целочисленное деление
a % b    # остаток от деления

Вывод на экран

print("Текст")
print("x =", x)
print(f"x = {x}, y = {y}")  # f-строка

Узнать тип

type(x)       # → <class 'int'>
type("abc")   # → <class 'str'>
⚠️ Не путай: = — присваивание (x = 5), == — сравнение (x == 5).

Изменение переменной

x = 10
x = x + 5   # x = 15
x += 3      # x = 18 (короткая запись)
x -= 2      # x = 16
x *= 2      # x = 32

Преобразование типов

int("42")        # → 42
float("3.14")    # → 3.14
str(123)         # → "123"
list("abc")      # → ['a', 'b', 'c']
bool(0)          # → False
bool(1)          # → True

Ввод с клавиатуры

name = input("Имя: ")
age = int(input("Возраст: "))

2. Условный оператор if-elif-else

Знаки сравнения

a == b   # равно
a != b   # не равно
a > b    # больше
a < b    # меньше
a >= b   # больше или равно
a <= b   # меньше или равно

Логические операторы

a and b   # И  (оба True)
a or b    # ИЛИ (хотя бы одно True)
not a     # НЕ  (переворачивает)

Синтаксис

if условие1:
    # код, если условие1 истинно
elif условие2:
    # код, если условие1 ложно, а условие2 истинно
else:
    # код, если все условия ложны
💡 Проверяется сверху вниз. Выполняется первый истинный блок, остальные пропускаются. 
age = 17
if age >= 18:
    print("Можно")
elif age >= 14:
    print("Можно с родителями")
else:
    print("Нельзя")

3. Циклы for и while

Цикл for

# По списку
for item in список:
    print(item)

# По range()
for i in range(5):        # 0..4
    print(i)

for i in range(2, 6):     # 2..5
    print(i)

for i in range(0, 10, 2): # 0,2,4,6,8
    print(i)

for i in range(5, 0, -1): # 5,4,3,2,1
    print(i)

Цикл while

while условие:
    # код, пока условие True

x = 5
while x > 0:
    print(x)
    x -= 1

Прерывание цикла

break      # выход из цикла
continue   # пропуск итерации

4. Строки

Срезы

s = "Python"
s[0]        # 'P'
s[-1]       # 'n'
s[0:3]      # 'Pyt' (0,1,2)
s[:3]       # 'Pyt'
s[3:]       # 'hon'
s[::-1]     # 'nohtyP' — переворот
s[::2]      # 'Pto' — каждый второй
len(s)      # 6

Основные методы

s.upper()           # "ПРИВЕТ"
s.lower()           # "привет"
s.strip()           # убрать пробелы
s.replace(old, new)
s.find(sub)        # индекс или -1
s.count(sub)       # количество
s.startswith(pref) # True/False
s.endswith(suf)    # True/False
s.isdigit()         # только цифры?
s.split()           # список слов
s.split(",")        # список по запятой
", ".join(list)     # список → строка
⚠️ Строки неизменяемы. Все методы возвращают новую строку.

5. Списки

Создание и доступ

lst = [1, 2, 3, 4, 5]
empty = []
mixed = [1, "текст", True]

lst[0]         # 1
lst[-1]        # 5
lst[1:3]       # [2, 3] — срез
len(lst)       # 5

Изменение

lst[0] = 99              # замена
lst.append(x)           # в конец
lst.insert(i, x)        # на позицию i
lst.remove(x)           # по значению
lst.pop(i)              # по индексу
lst.pop()               # последний
lst.extend([x, y])      # добавить список
lst.clear()             # очистить
lst2 = lst.copy()        # копия

Полезные функции

min(lst)          # минимум
max(lst)          # максимум
sum(lst)          # сумма (только числа)
sorted(lst)       # новый отсортированный
lst.sort()          # сортировка на месте
lst.sort(reverse=True)  # по убыванию
lst.count(x)        # количество вхождений
lst.index(x)        # индекс первого

Генератор списка (list comprehension)

[x*2 for x in range(5)]              # [0, 2, 4, 6, 8]
[x for x in range(10) if x % 2 == 0]  # [0, 2, 4, 6, 8]
[len(w) for w in слова]              # список длин слов

6. Словари

Создание и доступ

d = {"ключ": "значение", "x": 10}
empty = {}
d["новый"] = 5          # добавить
d["x"] = 20             # изменить
val = d["ключ"]         # получить (KeyError)
val = d.get("ключ")    # безопасно
val = d.get("ключ", 0) # с умолчанием

Перебор

# По ключам
for key in d:
    print(key, d[key])

# По парам
for key, val in d.items():
    print(key, "→", val)

# Только значения
for val in d.values():
    print(val)

# Проверка ключа
if key in d:
    print("Есть")

Основные методы

d.keys()            # все ключи
d.values()          # все значения
d.items()           # пары (ключ, значение)
d.pop(key, None)    # удалить и вернуть
d.update(d2)        # добавить пары из d2
len(d)              # количество пар

Подсчёт частоты (важнейший паттерн)

counter = {}
for item in data:
    counter[item] = counter.get(item, 0) + 1

# Ключ с максимальным значением
max_key = max(counter, key=counter.get)

7. Функции

Создание функции

def имя(параметр1, параметр2):
    """Описание."""
    результат = параметр1 + параметр2
    return результат

# Вызов
x = имя(5, 10)   # x = 15

Возврат нескольких значений

def stats(a, b):
    return a+b, a*b

s, p = stats(3, 5)  # s=8, p=15

Пустая функция

def заглушка():
    pass
💡 Локальные переменные внутри функции не видны снаружи. Используй параметры и return.

8. Чтение файлов

Чтение всего файла

with open("file.txt", "r",
          encoding="utf-8") as f:
    text = f.read()

Построчное чтение

with open("file.txt", "r",
          encoding="utf-8") as f:
    for line in f:
        line = line.strip()
        if line:
            print(line)

Другие методы чтения

f.read()         # весь файл одной строкой
f.readline()     # одна строка (или "" в конце)
f.readlines()    # список всех строк

Безопасное чтение

try:
    with open("file.txt", "r", encoding="utf-8") as f:
        data = f.read()
except FileNotFoundError:
    print("Файл не найден")
